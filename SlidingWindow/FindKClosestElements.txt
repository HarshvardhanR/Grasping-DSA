Leetcode 658: https://leetcode.com/problems/find-k-closest-elements/description/

Approach - Use two pointers to close in the gap until k 

class Solution {
    public List<Integer> findClosestElements(int[] arr, int k, int x) {
        int i = 0;
        int j = arr.length - 1;
        List<Integer> ans = new ArrayList<>();
        while((j - i + 1) > k){
            int iDiff = Math.abs(arr[i] - x);
            int jDiff = Math.abs(arr[j] - x);
            if(iDiff > jDiff){
                i++;
            }
            else{
                j--;
            } 
            
        }

        for(int p=i; p<=j; p++){
            ans.add(arr[p]);
        }

        return ans;
    }
}


Approach: Use Binary Search to get the answer

class Solution {
    public List<Integer> findClosestElements(int[] arr, int k, int x) {
        int left = 0;
        int right = arr.length - k;

        while (left < right) {
            int mid = (left + right) / 2;
            if (x - arr[mid] > arr[mid + k] - x) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }

        List<Integer> result = new ArrayList<>();
        for (int i = left; i < left + k; i++) {
            result.add(arr[i]);
        }

        return result;
    }
}
