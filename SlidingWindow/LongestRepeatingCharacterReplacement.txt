Leetcode 424: https://leetcode.com/problems/longest-repeating-character-replacement/description/

Keep a map to store the occurence and a count with a map to keep the window valid with k replacement 

class Solution {
    public int characterReplacement(String s, int k) {
        int[] freq = new int[26];
        int maxReplaced = 1;
        int p1 = 0;
        int p2 = 0;
        int currentMax = 0;

        while (p2 != s.length()) {
            int idx = s.charAt(p2) - 'A';
            freq[idx]++;
            currentMax = Math.max(currentMax, freq[idx]);

            while (p2 - p1 + 1 - currentMax > k) {
                freq[s.charAt(p1) - 'A']--;
                p1++;
            }

            maxReplaced = Math.max(maxReplaced, p2 - p1 + 1);
            p2++;
        }

        return maxReplaced;
    }
}
