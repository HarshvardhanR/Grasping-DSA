Leetcode 229: https://leetcode.com/problems/majority-element-ii/description/

Simply Use Boyer Moore
take two elements and its count set when met other decrease both counters.
Lastly loop over array again and set the count1 and count2
if it reaches n/3 just simply add it to answer


class Solution {
    public List<Integer> majorityElement(int[] nums) {
        Integer element1 = null, element2 = null;
        int count1 = 0, count2 = 0;

        for (int num : nums) {
            if (element1 != null && num == element1) {
                count1++;
            } else if (element2 != null && num == element2) {
                count2++;
            } else if (count1 == 0) {
                element1 = num;
                count1 = 1;
            } else if (count2 == 0) {
                element2 = num;
                count2 = 1;
            } else {
                count1--;
                count2--;
            }
        }

        count1 = 0;
        count2 = 0;

        for (int num : nums) {
            if (element1 != null && num == element1) count1++;
            else if (element2 != null && num == element2) count2++;
        }

        List<Integer> ans = new ArrayList<>();
        int n = nums.length;

        if (element1 != null && count1 > n / 3) {
            ans.add(element1);
        }
        if (element2 != null && count2 > n / 3) {
            ans.add(element2);
        }

        return ans;
    }
}
